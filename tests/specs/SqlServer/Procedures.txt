== should format a procedure ==
CREATE PROCEDURE dbo.GetUser @Id INT AS BEGIN SET NOCOUNT ON; SELECT * FROM dbo.Users WHERE Id = @Id; END
GO

[expect]
CREATE PROCEDURE dbo.GetUser @Id INT
AS
BEGIN
  SET NOCOUNT ON;
  SELECT *
  FROM dbo.Users
  WHERE Id = @Id;
END
GO

== should break parameters that don't fit ==
CREATE OR ALTER PROCEDURE dbo.GetOrders @CustomerId INT, @Since DATETIME2 = NULL, @IncludeCancelled BIT = 0, @Total MONEY OUTPUT AS
BEGIN

END
GO

[expect]
CREATE OR ALTER PROCEDURE dbo.GetOrders
  @CustomerId INT,
  @Since DATETIME2 = NULL,
  @IncludeCancelled BIT = 0,
  @Total MONEY OUTPUT
AS
BEGIN
  SELECT @Total = SUM(Total)
  FROM dbo.Orders
  WHERE CustomerId = @CustomerId;
END
GO

== should format a multi-statement table valued function ==
CREATE FUNCTION dbo.Numbers(@n INT) RETURNS @t TABLE (n INT) AS
BEGIN
DECLARE @i INT = 1;
WHILE @i <= @n BEGIN INSERT INTO @t VALUES (@i); SET @i += 1; END
RETURN;
END
GO

[expect]
CREATE FUNCTION dbo.Numbers(@n INT)
RETURNS @t TABLE (n INT)
AS
BEGIN
  DECLARE @i INT = 1;
  WHILE @i <= @n
  BEGIN
    INSERT INTO @t
    VALUES (@i);
    SET @i += 1;
  END
  RETURN;
END
GO

== should format a trigger ==
CREATE TRIGGER dbo.trg_Users ON dbo.Users AFTER INSERT, UPDATE AS
BEGIN
SET NOCOUNT ON;
IF UPDATE(Name) UPDATE u SET Modified = GETDATE() FROM dbo.Users u INNER JOIN inserted i ON i.Id = u.Id;
END
GO

[expect]
CREATE TRIGGER dbo.trg_Users ON dbo.Users AFTER INSERT, UPDATE
AS
BEGIN
  SET NOCOUNT ON;
  IF UPDATE(Name)
    UPDATE u
    SET Modified = GETDATE()
    FROM dbo.Users u
      INNER JOIN inserted i ON i.Id = u.Id;
END
GO
