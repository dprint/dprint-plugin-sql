== should format a plpgsql function and its body ==
create or replace function public.increment(i integer) returns integer language plpgsql as $$
declare result integer;
begin
result := i + 1;
return result;
end;
$$;

[expect]
create or replace function public.increment(i integer)
returns integer
language plpgsql
as $$
declare
  result integer;
begin
  result := i + 1;
  return result;
end;
$$;

== should format if statements ==
CREATE FUNCTION sign_of(x int) RETURNS text AS $$
BEGIN
IF x > 0 THEN RETURN 'positive'; ELSIF x < 0 THEN RETURN 'negative'; ELSE RETURN 'zero'; END IF;
END;
$$ LANGUAGE plpgsql;

[expect]
CREATE FUNCTION sign_of(x int)
RETURNS text
AS $$
BEGIN
  IF x > 0 THEN
    RETURN 'positive';
  ELSIF x < 0 THEN
    RETURN 'negative';
  ELSE
    RETURN 'zero';
  END IF;
END;
$$
LANGUAGE plpgsql;

== should format loops ==
CREATE FUNCTION loops() RETURNS void AS $body$
DECLARE r record; i int := 0;
BEGIN
LOOP i := i + 1; EXIT WHEN i > 10; END LOOP;
WHILE i > 0 LOOP i := i - 1; END LOOP;
FOR i IN 1..10 LOOP RAISE NOTICE '%', i; END LOOP;
FOR i IN REVERSE 10..1 BY 2 LOOP CONTINUE WHEN i = 5; END LOOP;
FOR r IN SELECT * FROM t WHERE t.active LOOP PERFORM do_something(r.id); END LOOP;
FOREACH i IN ARRAY ARRAY[1, 2, 3] LOOP NULL; END LOOP;
END;
$body$ LANGUAGE plpgsql;

[expect]
CREATE FUNCTION loops()
RETURNS void
AS $body$
DECLARE
  r record;
  i int := 0;
BEGIN
  LOOP
    i := i + 1;
    EXIT WHEN i > 10;
  END LOOP;
  WHILE i > 0 LOOP
    i := i - 1;
  END LOOP;
  FOR i IN 1..10 LOOP
    RAISE NOTICE '%', i;
  END LOOP;
  FOR i IN REVERSE 10..1 BY 2 LOOP
    CONTINUE WHEN i = 5;
  END LOOP;
  FOR r IN SELECT * FROM t WHERE t.active LOOP
    PERFORM do_something(r.id);
  END LOOP;
  FOREACH i IN ARRAY ARRAY[1, 2, 3] LOOP
    NULL;
  END LOOP;
END;
$body$
LANGUAGE plpgsql;

== should format exception handlers ==
CREATE FUNCTION safe_divide(a numeric, b numeric) RETURNS numeric AS $$
BEGIN
RETURN a / b;
EXCEPTION WHEN division_by_zero THEN RAISE NOTICE 'division by zero'; RETURN NULL;
WHEN OTHERS THEN RAISE;
END;
$$ LANGUAGE plpgsql;

[expect]
CREATE FUNCTION safe_divide(a numeric, b numeric)
RETURNS numeric
AS $$
BEGIN
  RETURN a / b;
EXCEPTION
  WHEN division_by_zero THEN
    RAISE NOTICE 'division by zero';
    RETURN NULL;
  WHEN OTHERS THEN
    RAISE;
END;
$$
LANGUAGE plpgsql;

== should format select into and get diagnostics ==
CREATE FUNCTION f() RETURNS int AS $$
DECLARE n int; c int;
BEGIN
SELECT count(*) INTO STRICT n FROM t;
UPDATE t SET a = 1;
GET DIAGNOSTICS c = ROW_COUNT;
RETURN n + c;
END;
$$ LANGUAGE plpgsql;

[expect]
CREATE FUNCTION f()
RETURNS int
AS $$
DECLARE
  n int;
  c int;
BEGIN
  SELECT count(*)
  INTO STRICT n
  FROM t;
  UPDATE t
  SET a = 1;
  GET DIAGNOSTICS c = ROW_COUNT;
  RETURN n + c;
END;
$$
LANGUAGE plpgsql;
