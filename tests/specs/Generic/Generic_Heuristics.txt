== should read brackets within an array as arrays ==
select percentile_disc(array[[null,1,0.5],[0.75,0.25,null]]) within group (order by x) from t;

[expect]
select percentile_disc(array[[null, 1, 0.5], [0.75, 0.25, null]]) within group (order by x)
from t;

== should not read an alias in a trigger as declarations ==
CREATE TRIGGER t AFTER UPDATE ON p REFERENCING OLD TABLE AS old_table FOR EACH STATEMENT EXECUTE PROCEDURE f(); CREATE TRIGGER u AFTER UPDATE ON p REFERENCING NEW TABLE AS new_table FOR EACH STATEMENT EXECUTE PROCEDURE f(); SELECT 1;

[expect]
CREATE TRIGGER t AFTER UPDATE ON p REFERENCING OLD TABLE AS old_table FOR EACH STATEMENT EXECUTE PROCEDURE f();
CREATE TRIGGER u AFTER UPDATE ON p REFERENCING NEW TABLE AS new_table FOR EACH STATEMENT EXECUTE PROCEDURE f();
SELECT 1;

== should not read the class of a function as declarations ==
CREATE FUNCTION f AS x USING FILE y;
SELECT 1;

[expect]
CREATE FUNCTION f
AS x USING FILE y;
SELECT 1;

== should not split an alter table at an action ==
ALTER TABLE t
TRUNCATE PARTITION p;

[expect]
ALTER TABLE t TRUNCATE PARTITION p;

== should read a hash within an operator as part of it ==
CREATE OPERATOR #%# (leftarg = int8, procedure = f);

[expect]
CREATE OPERATOR #%# (leftarg = int8, procedure = f);

== should read a hash at the start of a line as a comment ==
#=== a section ===
SELECT   1;

[expect]
#=== a section ===
SELECT 1;

== should read a hash after a statement as a comment ==
USE db; #inline
SELECT   1;

[expect]
USE db; #inline
SELECT 1;

== should not split a grant at its privileges ==
GRANT   SELECT ,   UPDATE   ON   t   TO   u
ALTER   TABLE   t   DROP   COLUMN   c

[expect]
GRANT SELECT, UPDATE ON t TO u
ALTER TABLE t DROP COLUMN c

== should split a statement after an alter table ==
ALTER TABLE t ALTER COLUMN d INT
SELECT * FROM t

[expect]
ALTER TABLE t ALTER COLUMN d INT
SELECT *
FROM t

== should keep a dot apart from a number that it's apart from ==
SELECT 1 . 5, t.a FROM t;

[expect]
SELECT 1 . 5, t.a
FROM t;

== should read a hash before a variable as a temporary table ==
CREATE TABLE #@temp (id INT);
SELECT a #@ b FROM #@temp;

[expect]
CREATE TABLE #@temp (id INT);
SELECT a #@ b
FROM #@temp;

== should keep a second query on the line of the first ==
SELECT 1 SELECT 2
INSERT INTO t (a) VALUES (2) SELECT 3

[expect]
SELECT 1 SELECT 2
INSERT INTO t (a)
VALUES (2) SELECT 3

== should split a second query that starts a line ==
SELECT   1
SELECT   2

[expect]
SELECT 1
SELECT 2

== should not split at a merge join hint that starts a line ==
SELECT * FROM a INNER HASH JOIN b ON a.id = b.id LEFT
MERGE JOIN c ON c.id = a.id

[expect]
SELECT *
FROM a
  INNER HASH JOIN b ON a.id = b.id
  LEFT MERGE JOIN c ON c.id = a.id

== should keep apart tokens that would read as one written together ==
SELECT @ a, @ @b, a . b FROM t;

[expect]
SELECT @ a, @ @b, a.b
FROM t;

== should split statements without semicolons after a create table ==
CREATE TABLE t (a INT)
INSERT INTO t VALUES (1)
DELETE FROM t

[expect]
CREATE TABLE t (a INT)
INSERT INTO t
VALUES (1)
DELETE FROM t

== should tell a routine from a statement naming a kind of routine ==
create trigger tr after insert on t for each row insert into log values (1);
create view v as select event, name from t;

[expect]
create trigger tr after insert on t for each row insert into log values (1);
create view v as
select event, name
from t;

== should not split at words that continue a statement on the next line ==
CREATE TEMPORARY TABLE t (a int)
ON COMMIT PRESERVE ROWS
SELECT * FROM t
USE INDEX (i)

[expect]
CREATE TEMPORARY TABLE t (a int) ON COMMIT PRESERVE ROWS
SELECT *
FROM t USE INDEX (i)
