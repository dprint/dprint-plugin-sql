== should indent joins beneath from ==
SELECT u.name, o.total FROM users u INNER JOIN orders o ON u.id = o.user_id WHERE o.total > 100

[expect]
SELECT u.name, o.total
FROM users u
  INNER JOIN orders o ON u.id = o.user_id
WHERE o.total > 100

== should format every kind of join ==
SELECT * FROM a JOIN b ON a.id = b.id INNER JOIN c ON c.id = a.id LEFT JOIN d ON d.id = a.id LEFT OUTER JOIN e ON e.id = a.id RIGHT JOIN f ON f.id = a.id RIGHT OUTER JOIN g ON g.id = a.id FULL JOIN h ON h.id = a.id FULL OUTER JOIN i ON i.id = a.id CROSS JOIN j;

[expect]
SELECT *
FROM a
  JOIN b ON a.id = b.id
  INNER JOIN c ON c.id = a.id
  LEFT JOIN d ON d.id = a.id
  LEFT OUTER JOIN e ON e.id = a.id
  RIGHT JOIN f ON f.id = a.id
  RIGHT OUTER JOIN g ON g.id = a.id
  FULL JOIN h ON h.id = a.id
  FULL OUTER JOIN i ON i.id = a.id
  CROSS JOIN j;
