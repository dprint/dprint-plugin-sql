== should put spaces around binary operators ==
SELECT a+b, a-b, a*b, a/b, a% b, a||b, a=b, a<>b, a!=b, a<b, a<=b, a>b, a>=b FROM t;

[expect]
SELECT a + b, a - b, a * b, a / b, a % b, a || b, a = b, a <> b, a != b, a < b, a <= b, a > b, a >= b
FROM t;

== should keep a percent sign written against both sides as it is ==
SELECT a%b, a % b FROM t;

[expect]
SELECT a%b, a % b
FROM t;

== should not put a space after a unary operator ==
SELECT -a, - b, +c, ~d, a * -1, a = -1, -(a + b), NOT a FROM t WHERE x = - 1;

[expect]
SELECT -a, -b, +c, ~d, a * -1, a = -1, -(a + b), NOT a
FROM t
WHERE x = -1;

== should tell unary from binary after keywords ==
SELECT CASE WHEN a THEN -1 ELSE -2 END, b - 1, NULL - 1 FROM t LIMIT -1;

[expect]
SELECT CASE WHEN a THEN -1 ELSE -2 END, b - 1, NULL - 1
FROM t
LIMIT -1;
