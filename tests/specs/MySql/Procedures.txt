== should format a procedure with a delimiter ==
DELIMITER //
CREATE PROCEDURE get_user(IN p_id INT, OUT p_name VARCHAR(100))
BEGIN
SELECT name INTO p_name FROM users WHERE id = p_id;
END //
DELIMITER ;

[expect]
DELIMITER //
CREATE PROCEDURE get_user(IN p_id INT, OUT p_name VARCHAR(100))
BEGIN
  SELECT name
  INTO p_name
  FROM users
  WHERE id = p_id;
END //
DELIMITER ;

== should format declarations and handlers ==
DELIMITER //
CREATE PROCEDURE p()
BEGIN
DECLARE done INT DEFAULT FALSE;
DECLARE v_id INT;
DECLARE cur CURSOR FOR SELECT id FROM t;
DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
DECLARE EXIT HANDLER FOR SQLEXCEPTION BEGIN ROLLBACK; RESIGNAL; END;
OPEN cur;
read_loop: LOOP
FETCH cur INTO v_id;
IF done THEN LEAVE read_loop; END IF;
END LOOP read_loop;
CLOSE cur;
END //
DELIMITER ;

[expect]
DELIMITER //
CREATE PROCEDURE p()
BEGIN
  DECLARE done INT DEFAULT FALSE;
  DECLARE v_id INT;
  DECLARE cur CURSOR FOR SELECT id FROM t;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    ROLLBACK;
    RESIGNAL;
  END;
  OPEN cur;
  read_loop: LOOP
    FETCH cur INTO v_id;
    IF done THEN
      LEAVE read_loop;
    END IF;
  END LOOP read_loop;
  CLOSE cur;
END //
DELIMITER ;

== should format control flow ==
DELIMITER //
CREATE PROCEDURE counter(IN n INT)
BEGIN
DECLARE i INT DEFAULT 0;
IF n < 0 THEN SET i = -1; ELSEIF n = 0 THEN SET i = 0; ELSE SET i = 1; END IF;
WHILE i < n DO SET i = i + 1; END WHILE;
REPEAT SET i = i - 1; UNTIL i <= 0 END REPEAT;
outer_loop: WHILE i < 10 DO SET i = i + 1; ITERATE outer_loop; END WHILE outer_loop;
CASE n WHEN 1 THEN SELECT 'one'; WHEN 2 THEN SELECT 'two'; ELSE SELECT 'many'; END CASE;
CASE WHEN n > 10 THEN SELECT 'big'; ELSE SELECT 'small'; END CASE;
END //
DELIMITER ;

[expect]
DELIMITER //
CREATE PROCEDURE counter(IN n INT)
BEGIN
  DECLARE i INT DEFAULT 0;
  IF n < 0 THEN
    SET i = -1;
  ELSEIF n = 0 THEN
    SET i = 0;
  ELSE
    SET i = 1;
  END IF;
  WHILE i < n DO
    SET i = i + 1;
  END WHILE;
  REPEAT
    SET i = i - 1;
  UNTIL i <= 0
  END REPEAT;
  outer_loop: WHILE i < 10 DO
    SET i = i + 1;
    ITERATE outer_loop;
  END WHILE outer_loop;
  CASE n
    WHEN 1 THEN
      SELECT 'one';
    WHEN 2 THEN
      SELECT 'two';
    ELSE
      SELECT 'many';
  END CASE;
  CASE
    WHEN n > 10 THEN
      SELECT 'big';
    ELSE
      SELECT 'small';
  END CASE;
END //
DELIMITER ;

== should format a trigger ==
DELIMITER //
CREATE TRIGGER before_insert BEFORE INSERT ON t FOR EACH ROW
BEGIN
IF NEW.a IS NULL THEN SET NEW.a = 0; END IF;
SET NEW.created_at = NOW();
END //
DELIMITER ;

[expect]
DELIMITER //
CREATE TRIGGER before_insert BEFORE INSERT ON t FOR EACH ROW
BEGIN
  IF NEW.a IS NULL THEN
    SET NEW.a = 0;
  END IF;
  SET NEW.created_at = NOW();
END //
DELIMITER ;

== should format a trigger with a single statement ==
CREATE   TRIGGER   t_ai   AFTER   INSERT   ON   t   FOR   EACH   ROW   INSERT   INTO   log   (id)   VALUES   (NEW.id);

[expect]
CREATE TRIGGER t_ai AFTER INSERT ON t FOR EACH ROW INSERT INTO log (id) VALUES (NEW.id);

== should format an event ==
CREATE EVENT cleanup ON SCHEDULE EVERY 1 DAY DO BEGIN DELETE FROM sessions WHERE expires_at < NOW(); END;

[expect]
CREATE EVENT cleanup ON SCHEDULE EVERY 1 DAY DO
BEGIN
  DELETE FROM sessions
  WHERE expires_at < NOW();
END;
