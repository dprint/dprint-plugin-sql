== should format a procedure ==
CREATE OR REPLACE PROCEDURE raise_salary (p_emp_id IN NUMBER, p_amount IN NUMBER) IS
v_salary NUMBER;
BEGIN
SELECT salary INTO v_salary FROM employees WHERE employee_id = p_emp_id;
UPDATE employees SET salary = v_salary + p_amount WHERE employee_id = p_emp_id;
COMMIT;
EXCEPTION WHEN NO_DATA_FOUND THEN DBMS_OUTPUT.PUT_LINE('not found'); WHEN OTHERS THEN ROLLBACK; RAISE;
END raise_salary;
/

[expect]
CREATE OR REPLACE PROCEDURE raise_salary (p_emp_id IN NUMBER, p_amount IN NUMBER) IS
  v_salary NUMBER;
BEGIN
  SELECT salary
  INTO v_salary
  FROM employees
  WHERE employee_id = p_emp_id;
  UPDATE employees
  SET salary = v_salary + p_amount
  WHERE employee_id = p_emp_id;
  COMMIT;
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    DBMS_OUTPUT.PUT_LINE('not found');
  WHEN OTHERS THEN
    ROLLBACK;
    RAISE;
END raise_salary;
/

== should format a function ==
CREATE OR REPLACE FUNCTION get_name (p_id IN NUMBER) RETURN VARCHAR2 IS v_name VARCHAR2(100); BEGIN SELECT name INTO v_name FROM t WHERE id = p_id; RETURN v_name; END get_name;
/

[expect]
CREATE OR REPLACE FUNCTION get_name (p_id IN NUMBER) RETURN VARCHAR2 IS
  v_name VARCHAR2(100);
BEGIN
  SELECT name
  INTO v_name
  FROM t
  WHERE id = p_id;
  RETURN v_name;
END get_name;
/

== should format loops ==
BEGIN
FOR i IN 1..10 LOOP DBMS_OUTPUT.PUT_LINE(i); END LOOP;
FOR r IN (SELECT * FROM employees) LOOP DBMS_OUTPUT.PUT_LINE(r.last_name); END LOOP;
FOR r IN c_emp LOOP NULL; END LOOP;
WHILE v_count < 10 LOOP v_count := v_count + 1; END LOOP;
LOOP FETCH c INTO v; EXIT WHEN c%NOTFOUND; END LOOP;
END;
/

[expect]
BEGIN
  FOR i IN 1..10 LOOP
    DBMS_OUTPUT.PUT_LINE(i);
  END LOOP;
  FOR r IN (SELECT * FROM employees) LOOP
    DBMS_OUTPUT.PUT_LINE(r.last_name);
  END LOOP;
  FOR r IN c_emp LOOP
    NULL;
  END LOOP;
  WHILE v_count < 10 LOOP
    v_count := v_count + 1;
  END LOOP;
  LOOP
    FETCH c INTO v;
    EXIT WHEN c%NOTFOUND;
  END LOOP;
END;
/

== should format bulk collect and forall ==
DECLARE TYPE t_ids IS TABLE OF NUMBER; v_ids t_ids;
BEGIN
SELECT id BULK COLLECT INTO v_ids FROM t;
FORALL i IN 1..v_ids.COUNT DELETE FROM u WHERE id = v_ids(i);
END;
/

[expect]
DECLARE
  TYPE t_ids IS TABLE OF NUMBER;
  v_ids t_ids;
BEGIN
  SELECT id
  BULK COLLECT INTO v_ids
  FROM t;
  FORALL i IN 1..v_ids.COUNT DELETE FROM u WHERE id = v_ids(i);
END;
/
