~~ dialect: oracle ~~
== should format quoted strings ==
SELECT q'[it's]', Q'{a}', q'!b!', nq'(c)' FROM dual;

[expect]
SELECT q'[it's]', Q'{a}', q'!b!', nq'(c)'
FROM dual;

== should format bind variables ==
SELECT * FROM t WHERE a = :a AND b = :1;

[expect]
SELECT *
FROM t
WHERE a = :a AND b = :1;

== should format outer joins ==
SELECT * FROM a, b WHERE a.id = b.id(+);

[expect]
SELECT *
FROM a, b
WHERE a.id = b.id(+);

== should format hierarchical queries ==
SELECT LEVEL, e.name FROM employees e START WITH e.manager_id IS NULL CONNECT BY PRIOR e.id = e.manager_id ORDER SIBLINGS BY e.name;

[expect]
SELECT LEVEL, e.name
FROM employees e
START WITH e.manager_id IS NULL
CONNECT BY PRIOR e.id = e.manager_id
ORDER SIBLINGS BY e.name;

== should format sequences ==
SELECT seq.NEXTVAL, seq.CURRVAL FROM dual;

[expect]
SELECT seq.NEXTVAL, seq.CURRVAL
FROM dual;

== should format sql plus commands ==
SET SERVEROUTPUT ON
SET DEFINE OFF
PROMPT Don't stop now
BEGIN DBMS_OUTPUT.PUT_LINE('hi'); END;
/
@other_script.sql
SPOOL off

[expect]
SET SERVEROUTPUT ON
SET DEFINE OFF
PROMPT Don't stop now
BEGIN
  DBMS_OUTPUT.PUT_LINE('hi');
END;
/
@other_script.sql
SPOOL off

== should format merge ==
MERGE INTO t USING s ON (t.id = s.id) WHEN MATCHED THEN UPDATE SET t.a = s.a WHEN NOT MATCHED THEN INSERT (id, a) VALUES (s.id, s.a);

[expect]
MERGE INTO t
USING s
ON (t.id = s.id)
WHEN MATCHED THEN
  UPDATE SET t.a = s.a
WHEN NOT MATCHED THEN
  INSERT (id, a) VALUES (s.id, s.a);

INSERT INTO t (a)
VALUES (1)
RETURNING id INTO v_id;
