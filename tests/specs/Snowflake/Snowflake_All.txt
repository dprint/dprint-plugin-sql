== should format a procedure and its body ==
CREATE OR REPLACE PROCEDURE p(x INT) RETURNS VARCHAR LANGUAGE SQL AS $$
DECLARE result VARCHAR;
BEGIN
IF (x > 0) THEN result := 'positive'; ELSE result := 'other'; END IF;
RETURN result;
END;
$$;

[expect]
CREATE OR REPLACE PROCEDURE p(x INT)
RETURNS VARCHAR
LANGUAGE SQL
AS $$
DECLARE
  result VARCHAR;
BEGIN
  IF (x > 0) THEN
    result := 'positive';
  ELSE
    result := 'other';
  END IF;
  RETURN result;
END;
$$;

== should format an anonymous block ==
DECLARE c INT DEFAULT 0;
BEGIN
FOR i IN 1 TO 10 DO c := c + i; END FOR;
RETURN c;
END;

[expect]
DECLARE
  c INT DEFAULT 0;
BEGIN
  FOR i IN 1 TO 10 DO
    c := c + i;
  END FOR;
  RETURN c;
END;

== should format semi-structured access ==
SELECT v:name::string, v:address.city, v['key'], f.value FROM t, LATERAL FLATTEN(input => t.v:items) f;

[expect]
SELECT v:name::string, v:address.city, v['key'], f.value
FROM t, LATERAL FLATTEN(input => t.v:items) f;

== should format qualify ==
SELECT * FROM t QUALIFY ROW_NUMBER() OVER (PARTITION BY a ORDER BY b) = 1;

[expect]
SELECT *
FROM t
QUALIFY ROW_NUMBER() OVER (PARTITION BY a ORDER BY b) = 1;

== should format match recognize ==
SELECT * FROM t MATCH_RECOGNIZE (PARTITION BY a ORDER BY b MEASURES FIRST(b) AS start_b ONE ROW PER MATCH PATTERN (up+ down+) DEFINE up AS price > LAG(price), down AS price < LAG(price));

[expect]
SELECT *
FROM t MATCH_RECOGNIZE (
  PARTITION BY a
  ORDER BY b
  MEASURES FIRST(b) AS start_b
  ONE ROW PER MATCH
  PATTERN (up+ down+)
  DEFINE up AS price > LAG(price), down AS price < LAG(price)
);
