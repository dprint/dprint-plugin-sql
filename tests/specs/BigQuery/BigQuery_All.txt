~~ dialect: bigquery ~~
== should format backtick paths ==
SELECT * FROM `project.dataset.table` WHERE _PARTITIONTIME > TIMESTAMP '2020-01-01';

[expect]
SELECT *
FROM `project.dataset.table`
WHERE _PARTITIONTIME > TIMESTAMP '2020-01-01';

== should format except and replace ==
SELECT * EXCEPT (a, b) REPLACE (c * 2 AS c) FROM t;

[expect]
SELECT * EXCEPT (a, b) REPLACE (c * 2 AS c)
FROM t;

== should format structs and arrays ==
SELECT STRUCT(1 AS a, 'x' AS b), [1, 2, 3], ARRAY<INT64>[1, 2], t.s.field FROM t, UNNEST(t.items) AS item;

[expect]
SELECT STRUCT(1 AS a, 'x' AS b), [1, 2, 3], ARRAY<INT64>[1, 2], t.s.field
FROM t, UNNEST(t.items) AS item;

== should format scripting ==
DECLARE x INT64 DEFAULT 0;
DECLARE y STRING;
SET x = x + 1;
IF x > 0 THEN SELECT 'positive'; ELSEIF x < 0 THEN SELECT 'negative'; ELSE SELECT 'zero'; END IF;
LOOP SET x = x + 1; IF x > 10 THEN BREAK; END IF; END LOOP;
WHILE x > 0 DO SET x = x - 1; END WHILE;
FOR row IN (SELECT 1 AS n) DO SELECT row.n; END FOR;

[expect]
DECLARE x INT64 DEFAULT 0;
DECLARE y STRING;
SET x = x + 1;
IF x > 0 THEN
  SELECT 'positive';
ELSEIF x < 0 THEN
  SELECT 'negative';
ELSE
  SELECT 'zero';
END IF;
LOOP
  SET x = x + 1;
  IF x > 10 THEN
    BREAK;
  END IF;
END LOOP;
WHILE x > 0 DO
  SET x = x - 1;
END WHILE;
FOR row IN (SELECT 1 AS n) DO
  SELECT row.n;
END FOR;

== should format blocks with exception handlers ==
BEGIN
SELECT 1 / 0;
EXCEPTION WHEN ERROR THEN
SELECT @@error.message;
END;

[expect]
BEGIN
  SELECT 1 / 0;
EXCEPTION
  WHEN ERROR THEN
    SELECT @@error.message;
END;

CREATE OR REPLACE PROCEDURE dataset.p(x INT64, OUT y INT64)

  SET y = x * 2;
END;

FROM t
WHERE TRUE
